cmake_minimum_required(VERSION 3.15)
project(rax_sdk_consumer LANGUAGES C CXX)
enable_testing()
find_package(rax REQUIRED CONFIG)
foreach(linkage shared static)
    if(linkage STREQUAL "shared")
        set(rax_target rax::rax)
    else()
        set(rax_target rax::rax_static)
    endif()
    if(NOT TARGET ${rax_target})
        message(FATAL_ERROR "Installed SDK is missing ${rax_target}")
    endif()
    foreach(example x86_64_basic cpp_engine cpp_registers cpp_fault_recovery cpp_instruction_info)
        if(example MATCHES "^cpp_")
            set(extension cpp)
        else()
            set(extension c)
        endif()
        set(consumer ${example}_${linkage})
        add_executable(${consumer} "../../examples/${example}.${extension}")
        target_compile_features(${consumer} PRIVATE c_std_11 cxx_std_17)
        target_link_libraries(${consumer} PRIVATE ${rax_target})
        if(WIN32 AND linkage STREQUAL "shared")
            add_custom_command(TARGET ${consumer} POST_BUILD
                COMMAND ${CMAKE_COMMAND} -E copy_if_different
                    "$<TARGET_FILE:rax::rax>" "$<TARGET_FILE_DIR:${consumer}>")
        endif()
        add_test(NAME ${consumer} COMMAND ${consumer})
    endforeach()
endforeach()
add_library(abi_check OBJECT ../analyze_abi.c ../fault_abi.c ../instruction_info_abi.c)
target_link_libraries(abi_check PRIVATE rax::rax_static)
target_compile_features(abi_check PRIVATE c_std_11)
if(MSVC)
    target_compile_options(abi_check PRIVATE /W4 /WX)
else()
    target_compile_options(abi_check PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()
