cmake_minimum_required(VERSION 3.20)
project(rax_native_code_integrity_policies LANGUAGES C)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The independent observer requires native Windows and MSVC")
endif()
add_executable(rax_native_code_integrity_policies probe-v1.c)
target_compile_features(rax_native_code_integrity_policies PRIVATE c_std_11)
target_link_libraries(rax_native_code_integrity_policies PRIVATE psapi)
target_compile_options(rax_native_code_integrity_policies PRIVATE /W4 /utf-8)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/probe-paths-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:rax_native_code_integrity_policies>\n")
