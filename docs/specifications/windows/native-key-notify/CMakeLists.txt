cmake_minimum_required(VERSION 3.20)
project(RaxNativeKeyNotify C)
if(NOT WIN32)
  message(FATAL_ERROR "Native registry notification controls require Windows")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(probe_names lifetime admission thread flags completion completion_stores)
set(probe_sources probe-lifetime-v1.c probe-admission-v2.c probe-thread-v3.c probe-flags-v4.c probe-completion-v5.c probe-completion-stores-v6.c)
set(probe_paths "")
foreach(name source IN ZIP_LISTS probe_names probe_sources)
  add_executable(notify_${name} ${source})
  target_compile_features(notify_${name} PRIVATE c_std_11)
  if(MSVC)
    target_compile_options(notify_${name} PRIVATE /W4 /WX)
  endif()
  target_link_libraries(notify_${name} PRIVATE ntdll kernel32 advapi32)
  string(APPEND probe_paths "$<TARGET_FILE:notify_${name}>\n")
endforeach()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/probe-paths-$<CONFIG>.txt" CONTENT "${probe_paths}")
