cmake_minimum_required(VERSION 3.20)
project(rax_native_file_read_control LANGUAGES C)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Controls require Windows/MSVC-compatible targets")
endif()
set(CMAKE_C_STANDARD_LIBRARIES "")
foreach(kind scalars transitions)
  add_executable(file_read_${kind} ${kind}-v1.c)
  set_target_properties(file_read_${kind} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES MSVC_RUNTIME_LIBRARY MultiThreaded)
  target_compile_options(file_read_${kind} PRIVATE /O2 /W4 /WX /GS- /Zl)
  target_link_options(file_read_${kind} PRIVATE /nodefaultlib /entry:entry /subsystem:console)
  target_link_libraries(file_read_${kind} PRIVATE kernel32)
endforeach()
