cmake_minimum_required(VERSION 3.20)
project(rax_current_version_control LANGUAGES C)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The controls require a Windows/MSVC-compatible target")
endif()
set(CMAKE_C_STANDARD_LIBRARIES "")
add_executable(current_version_profile probe-relative-v1.c)
set_target_properties(current_version_profile PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_options(current_version_profile PRIVATE /O2 /W4 /WX /GS- /Zl)
target_link_options(current_version_profile PRIVATE /nodefaultlib /entry:entry /subsystem:console)
target_link_libraries(current_version_profile PRIVATE kernel32)

add_executable(current_version_parent probe-parent-v1.c)
set_target_properties(current_version_parent PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_options(current_version_parent PRIVATE /O2 /W4 /WX /GS- /Zl)
target_link_options(current_version_parent PRIVATE /nodefaultlib /entry:entry /subsystem:console)
target_link_libraries(current_version_parent PRIVATE kernel32)

add_executable(current_version_physical probe-physical-v1.c)
set_target_properties(current_version_physical PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_options(current_version_physical PRIVATE /O2 /W4 /WX /GS- /Zl)
target_link_options(current_version_physical PRIVATE /nodefaultlib /entry:entry /subsystem:console)
target_link_libraries(current_version_physical PRIVATE kernel32)
