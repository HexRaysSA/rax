cmake_minimum_required(VERSION 3.20)
project(rax_native_file_data_section_controls LANGUAGES CXX)

# Freestanding controls import Kernel32 and one SEH entry explicitly. CMake's
# default GUI/printing libraries are not inputs to these native observations.
set(CMAKE_CXX_STANDARD_LIBRARIES "")

if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "These independent native Windows controls require MSVC-compatible SEH")
endif()
file(REAL_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." checkout)
file(REAL_PATH "${CMAKE_BINARY_DIR}" binary)
cmake_path(IS_PREFIX checkout "${binary}" NORMALIZE contained)
if(NOT contained)
  message(FATAL_ERROR "Keep the build and its file fixtures inside this checkout")
endif()

if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  set(machine x86)
  set(seh_dll msvcrt.dll)
  set(seh_symbol _except_handler3)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|arm64|aarch64|AARCH64)$"
    OR CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
  set(machine arm64)
  set(seh_dll ntdll.dll)
  set(seh_symbol __C_specific_handler)
else()
  set(machine x64)
  set(seh_dll ntdll.dll)
  set(seh_symbol __C_specific_handler)
endif()
set(seh_def "${CMAKE_CURRENT_BINARY_DIR}/seh.def")
set(seh_lib "${CMAKE_CURRENT_BINARY_DIR}/seh.lib")
file(WRITE "${seh_def}" "LIBRARY ${seh_dll}\nEXPORTS\n${seh_symbol}\n")
add_custom_command(OUTPUT "${seh_lib}"
  COMMAND "${CMAKE_AR}" "/def:${seh_def}" "/out:${seh_lib}" "/machine:${machine}"
  DEPENDS "${seh_def}" VERBATIM)
add_custom_target(seh_import DEPENDS "${seh_lib}")

enable_testing()
foreach(probe scalar coherence details transitions query status post-copy share)
  if(probe STREQUAL "coherence")
    set(version 2)
  else()
    set(version 1)
  endif()
  add_executable("file-data-section-${probe}" "native-file-data-section-${probe}-v${version}.cpp")
  add_dependencies("file-data-section-${probe}" seh_import)
  target_compile_features("file-data-section-${probe}" PRIVATE cxx_std_17)
  target_compile_options("file-data-section-${probe}" PRIVATE /O2 /W4 /GS- /Zl /utf-8)
  if(probe MATCHES "^(coherence|details|transitions|query|status)$")
    target_compile_options("file-data-section-${probe}" PRIVATE /EHa)
    if(CMAKE_SIZEOF_VOID_P EQUAL 4)
      target_sources("file-data-section-${probe}" PRIVATE native-probe-x86-seh-v1.cpp)
    endif()
  endif()
  target_link_options("file-data-section-${probe}" PRIVATE /nodefaultlib /entry:entry /subsystem:console)
  target_link_libraries("file-data-section-${probe}" PRIVATE kernel32 "${seh_lib}")
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${probe}")
  add_test(NAME "native-file-data-section-${probe}" COMMAND "$<TARGET_FILE:file-data-section-${probe}>")
  set_tests_properties("native-file-data-section-${probe}" PROPERTIES
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${probe}")
endforeach()
